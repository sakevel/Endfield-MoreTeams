cmake_minimum_required(VERSION 3.25)
project(EndfieldMoreTeams VERSION 0.1.21 LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Windows x64 required")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_compile_options(/utf-8 /W4 /permissive-)
add_compile_definitions(UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
add_library(MoreTeams SHARED src/plugin.cpp)
target_include_directories(MoreTeams PRIVATE src sdk/include)
set_target_properties(MoreTeams PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/package/$<CONFIG>/more-teams")
add_custom_target(package ALL DEPENDS MoreTeams)
foreach(f mod.ini icon.png controller.lua native.lua marks.lua ui.lua zml-package.json)
  add_custom_command(TARGET package POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy "${CMAKE_CURRENT_SOURCE_DIR}/mod/${f}" "${CMAKE_BINARY_DIR}/package/$<CONFIG>/more-teams/${f}")
endforeach()
enable_testing()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(ZML_LUPA_PATH "" CACHE PATH "Existing Lupa installation")
set(ZML_CLIENT_LUA "" CACHE PATH "Freshly extracted client fixtures")
set(ZML_INSTALL_SCRIPT "" CACHE FILEPATH "Generic installer, disposable copy tests")
add_executable(ContractTests tests/contract.cpp)
target_include_directories(ContractTests PRIVATE src sdk/include)
if(ZML_CLIENT_LUA)
  add_test(NAME client_contract COMMAND ContractTests "$<TARGET_FILE:MoreTeams>" "${ZML_CLIENT_LUA}" "${CMAKE_BINARY_DIR}/client-patched")
endif()
if(ZML_LUPA_PATH)
  add_test(NAME behavior COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/behavior.py" "${ZML_LUPA_PATH}")
  if(ZML_CLIENT_LUA)
    add_test(NAME native_scenarios COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/native.py" "${ZML_LUPA_PATH}" "${CMAKE_BINARY_DIR}/client-patched" "${ZML_CLIENT_LUA}")
    set_tests_properties(native_scenarios PROPERTIES DEPENDS client_contract)
    add_test(NAME native_ui COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/ui.py" "${ZML_LUPA_PATH}" "${CMAKE_BINARY_DIR}/client-patched/ctrl.lua" "${ZML_CLIENT_LUA}/ui/panels/navitargetactionmenu/navitargetactionmenuctrl.lua")
    set_tests_properties(native_ui PROPERTIES DEPENDS client_contract)
  endif()
endif()
add_test(NAME package_manifest COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/package.py" "${CMAKE_BINARY_DIR}/package/$<CONFIG>/more-teams")
add_test(NAME mark_assets COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/marks.py")
if(ZML_INSTALL_SCRIPT)
  add_test(NAME install_rollback COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/install.py" "${ZML_INSTALL_SCRIPT}" "${CMAKE_BINARY_DIR}/package/$<CONFIG>/more-teams")
endif()

set(ZML_SERVICES_EXE "" CACHE FILEPATH "Optional actual Loader services fixture")
if(ZML_SERVICES_EXE)
  add_test(NAME public_services COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/services.py" "${ZML_SERVICES_EXE}" "${CMAKE_BINARY_DIR}/package/$<CONFIG>/more-teams")
endif()
add_test(NAME local_storage COMMAND powershell -NoProfile -ExecutionPolicy Bypass -File "${CMAKE_CURRENT_SOURCE_DIR}/tests/storage.ps1")
